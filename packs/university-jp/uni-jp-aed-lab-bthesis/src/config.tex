	urlcolor=magenta,
% colorlinks=false の場合のテクスト枠色
	linkbordercolor={1 0 0},

	citebordercolor={0 1 0},
% colorlinks=false の場合のURL枠色
	urlbordercolor={0 1 1},
% 紙サイズ変更オプション, falseを推奨
        setpagesize=false
