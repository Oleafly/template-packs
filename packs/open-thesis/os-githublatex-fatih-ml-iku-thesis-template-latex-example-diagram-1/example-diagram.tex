\documentclass[tikz,border=6pt]{standalone}
\usetikzlibrary{arrows.meta,positioning}
\begin{document}
\begin{tikzpicture}[
    node distance=14mm,
    box/.style={draw,rounded corners,minimum width=26mm,minimum height=10mm,align=center},
    >={Stealth[]}]
  \node[box] (in)  {Input};
  \node[box,right=of in]  (proc) {Process};
  \node[box,right=of proc] (out) {Output};
  \draw[->] (in) -- (proc);
  \draw[->] (proc) -- (out);
\end{tikzpicture}
\end{document}
