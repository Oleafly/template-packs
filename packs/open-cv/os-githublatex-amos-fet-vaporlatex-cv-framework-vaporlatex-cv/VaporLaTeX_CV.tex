% --- SHADOW SELECTOR ---
\newcommand{\drawvaporshadow}[4]{
	% #1: choice, #2: color, #3: rounded edges, #4: coordinates
	\def\vpchoice{#1}
	
	\ifdefstring{\vpchoice}{random}{%
		\pgfmathrandominteger{\randpattern}{0}{4}
		\ifcase\randpattern
		\def\vpchoice{dots}\or
		\def\vpchoice{crosshatch}\or
		\def\vpchoice{grid}\or
		\def\vpchoice{solid}\or
		\def\vpchoice{lines}\fi
	}{}
	
	\ifdefstring{\vpchoice}{dots}{
		\fill[#2, fill opacity=0.1, #3] #4;
		\fill[pattern={Dots[angle=45, distance=1mm, radius=1pt]}, pattern color=#2, #3] #4;
	}{}
	\ifdefstring{\vpchoice}{crosshatch}{
		\fill[#2, fill opacity=0.1, #3] #4;
		\fill[pattern={Lines[angle=45, distance=1mm, line width=0.5pt]}, pattern color=#2, #3] #4;
		\fill[pattern={Lines[angle=-45, distance=1mm, line width=0.5pt]}, pattern color=#2, #3] #4;
	}{}
	\ifdefstring{\vpchoice}{grid}{
		\fill[#2, fill opacity=0.1, #3] #4;
		\fill[pattern={Lines[angle=0, distance=1.25mm, line width=0.5pt]}, pattern color=#2, #3] #4;
		\fill[pattern={Lines[angle=90, distance=1.25mm, line width=0.5pt]}, pattern color=#2, #3] #4;
	}{}
	\ifdefstring{\vpchoice}{solid}{
		\fill[#2, #3] #4;
	}{}
	\ifdefstring{\vpchoice}{lines}{
		\fill[#2, fill opacity=0.1, #3] #4;
		\fill[pattern={Lines[angle=-45, distance=1mm, line width=0.5pt]}, pattern color=#2, #3] #4;
	}{}
}

	-- ([xshift=-3mm]interior.north east)

	-- ([yshift=3mm]interior.south west) [rounded corners=2mm]

	([yshift=3mm]interior.south west) 
	-- ([xshift=-3mm, yshift=3mm]interior.south east)

	-- (interior.north east) [rounded corners=1.35mm]   
	-- (interior.south east) 
	-- (interior.south west) [sharp corners]          
	-- cycle;                
	
	% Left scroll button
	\fill[vpThemeColor!30]
	(interior.south west) 
	-- ([yshift=3mm]interior.south west)
	-- ([xshift=3mm, yshift=3mm]interior.south west)

	% Up scroll button

	([xshift=-3mm]interior.north east)

	([xshift=-3mm, yshift=6mm]interior.south east)

	% Inner borders and dividers
	\draw[vpThemeColor, line width=0.75pt]
	([yshift=3mm]interior.south west) 
	-- ([xshift=-3mm, yshift=3mm]interior.south east)
	-- ([xshift=-3mm]interior.north east)
	
	([xshift=3mm, yshift=3mm]interior.south west) node (h1) {} 
	-- ([xshift=3mm]interior.south west) 
	
	([xshift=-3mm, yshift=3mm]interior.south east) 
	-- ([xshift=-3mm]interior.south east) 
	
	([xshift=-6mm, yshift=3mm]interior.south east) node (h2) {} 
	-- ([xshift=-6mm]interior.south east) 
	
	([xshift=-3mm, yshift=3mm]interior.south east) 
	-- ([yshift=3mm]interior.south east)
	
	([xshift=-3mm, yshift=6mm]interior.south east) node (v2) {}
	-- ([yshift=6mm]interior.south east)
	
	([xshift=-3mm, yshift=-3mm]interior.north east) node (v1) {}
	-- ([yshift=-3mm]interior.north east);
	
	% --- Vertical scrollbar thumb ---
	\path (v1);
	\pgfgetlastxy{\xVuno}{\yVuno}
	\path (v2);
	\pgfgetlastxy{\xVdue}{\yVdue}
	\pgfmathsetmacro{\distV}{abs(\yVdue-\yVuno)}
	
	\ifdim\distV pt > 15mm\relax
	\pgfmathsetmacro{\rectHig}{\distV * (0.20 + 0.10*rnd)}
	\pgfmathsetmacro{\rectPosV}{rnd*(\distV-\rectHig)}
	
	\fill[vpThemeColor!30] 
	([yshift=\rectPosV pt]v2.center) rectangle ++(3mm, \rectHig pt);
	
	\draw[vpThemeColor, line width=0.75pt]
	([yshift=\rectPosV pt]v2.center) -- ++(0,\rectHig pt)
	([yshift=\rectPosV pt]v2.center) -- ++(3mm,0)
	([yshift=\rectPosV pt+\rectHig]v2.center) -- ++(3mm,0)
	node[scale=0.9, text=vpThemeColor] at([xshift=1.6mm,yshift=\rectPosV pt+\rectHig/2+1.1mm]\xVuno,\yVdue) {\faIcon{angle-double-up}}
	node[scale=0.9, text=vpThemeColor] at([xshift=1.6mm,yshift=\rectPosV pt+\rectHig/2-1.1mm]\xVuno,\yVdue) {\faIcon{angle-double-down}};
	\fi
	
	% --- Horizontal scrollbar thumb ---
	\path (h1);
	\pgfgetlastxy{\xHuno}{\yHuno}
	\path (h2);
	\pgfgetlastxy{\xHdue}{\yHdue}
	\pgfmathsetmacro{\distH}{\xHdue-\xHuno}
	
	\ifdim\distH pt > 15mm\relax
	\pgfmathsetmacro{\rectLen}{\distH * (0.10 + 0.25*rnd)}
	\pgfmathsetmacro{\rectPosH}{rnd*(\distH-\rectLen)}
	
	\fill[vpThemeColor!30] 
	([xshift=\rectPosH pt]h1.center) rectangle ++(\rectLen pt, -3mm);
	
	\draw[vpThemeColor, line width=0.75pt]
	([xshift=\rectPosH pt]h1.center) -- ++(\rectLen pt,0)
	([xshift=\rectPosH pt]h1.center) -- ++(0,-3mm)
	([xshift=\rectPosH pt+\rectLen]h1.center) -- ++(0,-3mm)
	node[scale=0.9, text=vpThemeColor] at([xshift=\rectPosH pt+\rectLen/2-1.1mm]\xHuno,1.9mm) {\faIcon{angle-double-left}}
	node[scale=0.9, text=vpThemeColor] at([xshift=\rectPosH pt+\rectLen/2+1.1mm]\xHuno,1.9mm) {\faIcon{angle-double-right}};
	\fi
}

		outer sep=0pt
		]
		% 1. GHOST NODES

		(N1) at (0,0) {\phantom{\Huge \textbf{\MakeUppercase{#2}}}};
		\node[inner xsep=10pt, inner ysep=8pt, outer sep=0pt, anchor=north west] 
		(N2) at ([yshift=8pt]N1.south west) {\phantom{\Huge \textbf{\MakeUppercase{#3}}}};
		
		% 2. SHADOW
		\drawvaporshadow{#5}{#1}{rounded corners=2.5mm}{
			([xshift=1.5mm, yshift=-2mm]N1.north west) --
			([xshift=1.5mm, yshift=-2mm]N2.south west) --
			([xshift=1.5mm, yshift=-2mm]N2.south east) |-   
			([xshift=1.5mm, yshift=-2mm]N1.south east) --
			([xshift=1.5mm, yshift=-2mm]N1.north east) -- cycle
		}
		
		% 3. BOX 
		\fill[#1, rounded corners=2.5mm]
		(N1.north west) --
		(N2.south west) --
		(N2.south east) |-   
		(N1.south east) --
		(N1.north east) -- cycle;
		
		% 4. TEXT
		\node[inner sep=0pt, outer sep=0pt, text=white] at (N1.center) {\Huge \textbf{\MakeUppercase{#2}}};
		\node[inner sep=0pt, outer sep=0pt, text=white] at (N2.center) {\Huge \textbf{\MakeUppercase{#3}}};
		
		% 5. ROLE

			\node[anchor=north west, inner sep=0pt, text=#1, font=\sffamily\large, yshift=-8pt] 
			at ([xshift=2pt, yshift=-3pt]N2.south west) {\MakeUppercase{#4}};
		}
	\end{tikzpicture}
	\par\vspace{1cm}
}

		\fill[pattern=checkerboard, pattern color=vpThemeColor!10, rounded corners=1.25mm] 
		([xshift=2pt, yshift=2pt]frame.south west) 
		rectangle 
		([xshift=-2pt, yshift=-2pt]frame.north east);
		
		\path (interior.south west); \pgfgetlastxy{\xSW}{\ySW}
		\path (interior.north east); \pgfgetlastxy{\xNE}{\yNE}
		
		\pgfmathsetmacro{\lw}{\xNE - \xSW} 
		\pgfmathsetmacro{\colW}{(\lw - 22.7622 - (#4-1)*17.0716)/#4} 
		\pgfmathsetmacro{\panelW}{\colW + 11.3811} 
		
		\foreach \i in {1,...,#4} {
			\pgfmathsetmacro{\startX}{\xSW + 5.6905 + (\i-1)*(\colW + 17.0716)}
			\pgfmathsetmacro{\endX}{\startX + \panelW}
			\pgfmathsetmacro{\startY}{\ySW + 5.6905}
			\pgfmathsetmacro{\endY}{\yNE - 5.6905}
			
			\draw[vpThemeColor!30, line width=1pt, fill=white, rounded corners=1.25mm] 
			(\startX pt, \startY pt) rectangle (\endX pt, \endY pt);
		}
	}
}
