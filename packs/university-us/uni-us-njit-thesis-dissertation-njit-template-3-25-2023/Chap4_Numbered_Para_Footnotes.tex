  	\end{singlespace}
 \end{enumerate}

 \begin{itemize}
 	 	\begin{singlespace}
	 \item First sentence First sentence First sentence First sentence First sentence First sentence First sentence First sentence First sentence.  
	 
	 \item Second sentence.
	 
	 \item Third sentence.
	 
  	\end{singlespace}
\end{itemize}
