% Single-space multi-line chapter titles; bold ALL-CAPS appendix headings.
\makeatletter
\def\@makechapterhead#1{%
  \begin{center}%
    \my@addappropriatespace%
    \ifmy@appendix
      \textls*[25]{\textbf{\MakeUppercase{\@chapapp}\enspace\thechapter}}%
      \par\vspace{\baselineskip}%
      \begin{singlespace}{\large\bfseries\MakeUppercase{#1}}\end{singlespace}%
    \else
      \textls*[25]{\textbf{\MakeUppercase{\@chapapp}\enspace\thechapter}}%
      \par\vspace{\baselineskip}%
      \begin{singlespace}{\large\bfseries #1}\end{singlespace}%
    \fi
    \par
  \end{center}%
}
\makeatother

% Footnote: begin directly below the solid line.
% \footnoterule: no trailing kern; \footnotesep: small strut so the first line
% sits immediately below the rule (cls default is \baselineskip which leaves a
% full empty line above the footnote text).
\setlength{\footnotesep}{19pt}
\renewcommand{\footnoterule}{%
  \kern-3pt
  \hrule width 2in height 0.4pt
  \kern-8pt%
}

% Bibliography: single-spaced within entry, one double-spaced blank line between entries.
% natbib exposes \bibsep for this; cls already sets \singlespacing inside thebibliography.

\renewcommand{\bibfont}{\singlespacing}
