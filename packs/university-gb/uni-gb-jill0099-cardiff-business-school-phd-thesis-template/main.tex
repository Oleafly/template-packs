% Policy Section 6.3: characters in all other text -- footnotes, figure
% captions and the like -- "must not be less than 11 point". Section 6.5
% extends the same requirement to characters inside tables and figures.
%
% At a 12pt base, LaTeX's \footnotesize (10pt), \scriptsize (8pt) and
% \tiny (6pt) all fall below that floor, and table notes and stars
% conventionally use them. Redefining them to 11pt keeps the document
% compliant even where a chapter reaches for a smaller size. \small is
% already 11pt at this base and is left alone.
%
% If you find yourself wanting something smaller to make a wide table
% fit, the compliant fixes are a landscape page (\begin{landscape}),
% splitting the table, or dropping columns -- not shrinking the type.
%TC:macro \footnote [ignore]
\usepackage[hang,flushmargin]{footmisc}
\renewcommand{\footnotesize}{\fontsize{11pt}{13pt}\selectfont}
\renewcommand{\scriptsize}{\fontsize{11pt}{13pt}\selectfont}
\renewcommand{\tiny}{\fontsize{11pt}{13pt}\selectfont}
\captionsetup{font={stretch=1.0},labelfont=bf,size=normalsize}

% Mathematical superscripts and subscripts are necessarily smaller than
% body text -- no thesis sets an index at 11pt -- so the 11pt floor above
% cannot apply to them. But LaTeX's defaults are smaller than they need
% to be (8.4pt and 6pt at a 12pt base), and Section 6.5 asks that
% characters inside tables and figures be "legible and of a sufficiently
% large font size for general accessibility" -- which is precisely the
% case of significance stars in a regression table. These raise the
% script levels as far as the notation allows.
%
% Verify what actually ends up in your PDF with tools/checkfonts.py.
% unicode-math re-derives the script levels from the maths font's own
% OpenType constants when the document begins, which silently undoes a
% plain \DeclareMathSizes for the body size. Deferring it wins.
\AtBeginDocument{%
  \DeclareMathSizes{12}{12}{10}{9}%
  \DeclareMathSizes{11}{11}{9.5}{9}%
}

% WHERE TABLES AND FIGURES GO
%
% The policy is silent on placement. Section 6.4 only requires that
% tables, figures and appendices stay inside the single page-number
% sequence, which they do either way. So this is convention -- and the
% convention differs between a thesis and a journal submission.
%
% A THESIS is the final artefact. Nobody typesets it afterwards, and
% your examiners read exactly what you submit, so the exhibits stay in
% the chapter. In empirical finance they conventionally take a float
% page of their own -- \begin{table}[p] -- because a regression table
% rarely fits beside the text discussing it. That is what the worked
% tables in chapters/chapter1.tex do.
%
% A JOURNAL SUBMISSION is the opposite. JF, JFE, RFS and JFQA want
% "[Table 1 about here]" in the text with the exhibits collected at the
% end, because their typesetters place the floats. When you cut a
% chapter into a submission, uncomment the two lines below rather than
% moving anything by hand: endfloat moves every float to the end and
% leaves the markers behind for you.
%
%   \usepackage[tablesfirst]{endfloat}
%   \renewcommand{\efloatseparator}{\mbox{}}
%
% Use \FloatBarrier (from placeins, already loaded) to stop a float
% drifting past a section heading it belongs to.
