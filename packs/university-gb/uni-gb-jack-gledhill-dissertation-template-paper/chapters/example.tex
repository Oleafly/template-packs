% This is an example of a chapter.
% Each chapter should have a title describing its contents, and a label can link back to it elsewhere in the paper.

% Within a chapter, you can have as many sections as you like.
% Each of those sections can have as many subsections as you like.
% In both cases, you can disable the numbering by adding a * after the \section or \subsection command,
% but this will also remove it from the ToC so use it sparingly.

\chapter{Example Chapter}
\label{chapter:example}

\section{Section 1 of Chapter 1}
\label{section:example}

\subsection{Subsection 1 of Section 1 of Chapter 1}
\label{subsection:example}
