%Diese Datei enthält eigene shortcuts die du beliebig anpassen kannst


%Erzeugt den Text "vgl. <String>, <Kapitelnummer>", wobei der erste String einfach
%übernommen wird, der zweite Wert soll das label des Kapitels sein
%"\vgl{}{}" erzeugt kleines "vgl.", "\Vgl{}{}" erzeugt großes "Vgl."
%Beispiel:
%   "...wie in der Einleitung erwähnt, \vgl{Einleitung}{sec:intro}"
%erzeugt:
%   "...wie in der Einleitung erwähnt, vgl. Einleitung, Kapitel 1.1"
\newcommand{\vgl}[2]{vgl. #1, Kapitel \ref{#2}}
\newcommand{\Vgl}[2]{Vgl. #1, Kapitel \ref{#2}}
