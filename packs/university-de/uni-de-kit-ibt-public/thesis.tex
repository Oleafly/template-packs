\input{Template/ibtThesis}      % This file defines our template based on the memoir class. Normally, you shouldn't need to change anything. If you want to make adjustments or are wondering about a particular behavior, this might be the first place to look at, though.

\setsecnumdepth{subsubsection}    % indenture levels included in the table of contents

\input{Template/commands}       % Have a look at this file regarding convenience commands

\ifthesisIsDiss
	\preto\chapter\acresetall % expand acronyms once in each chapter
\fi

\include{abbreviations}			% Here you can define your abbreviations

\begin{document}
\pagestyle{fancy}

\ifthesisIsDiss  %Title
  \input{titleDiss}
  \input{License/license}
  \thispagestyle{empty}
\else
  \include{title}
\fi
\frontmatter

\rmfamily
\addtocontents{toc}{\protect\thispagestyle{empty}}
\listoftodos
\include{Parts/00-todos}
\clearemptydoublepage

\include{Parts/01-abstract}
\clearemptydoublepage

% German "Zusammenfassung" only required for PhD Thesis or when written in German. 
% Toggle in configuration block

\ifthesisIsDiss

\else
  \ifenglish
  % Do nothing, as no "Zusammenfassung" required
  \else
    \include{Parts/02-zusammenfassung}
    \clearemptydoublepage
  \fi
\fi

\include{Parts/03-acknowledgments}
\clearemptydoublepage
\tableofcontents*
\clearemptydoublepage

\chapter*{Abbreviations}

\addcontentsline{toc}{chapter}{Abbreviations}
\markboth{Abbreviations}{Abbreviations}
\printacronyms[heading=none]
\clearemptydoublepage

\chapterstyle{madsen}
\ifafive  % right margin too small to have chapter numbers there
  \renewcommand*{\printchapternum}{%
    \hspace{0.4em}%
      \resizebox{!}{4ex}{%
        \chapnamefont\bfseries\sffamily\thechapter}%
  }%
\fi
\acresetall     % reintroduce any abbreviations potentially used in the abstract
\mainmatter
\input{Parts/04-intro}
\clearemptydoublepage

\ifthesisIsDiss  %Title

\fi

\input{Parts/05-medicalFundamentals}

\ifthesisIsDiss  %Title
\part{Methods}
\fi

\input{Parts/07-fancyMethodA}
\clearemptydoublepage


\ifthesisIsDiss  %Title
\part{Results}
\fi

\input{Parts/08-results}
\clearemptydoublepage

\input{Parts/09-discussion}

\input{Parts/10-outlook}
\clearemptydoublepage
\input{Parts/11-conclusion}
\clearemptydoublepage

\begin{appendix}

%\addtocontents{toc}{\protect\setcounter{tocdepth}{0}}
